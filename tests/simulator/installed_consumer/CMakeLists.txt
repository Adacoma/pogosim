cmake_minimum_required(VERSION 3.16)
project(pogosim_installed_consumer LANGUAGES C CXX)
set(CMAKE_C_STANDARD 11)
set(CMAKE_CXX_STANDARD 20)
# Reuse dependency discovery modules, but never source-tree simulator headers.
list(APPEND CMAKE_MODULE_PATH "${SOURCE_DIR}/cmake/modules")
find_package(SDL2 REQUIRED)
find_package(SDL2_gfx REQUIRED)
find_package(SDL2_ttf REQUIRED)
find_package(yaml-cpp REQUIRED)
find_package(spdlog REQUIRED)
find_package(fmt REQUIRED)
find_package(Arrow REQUIRED)
if(TARGET yaml-cpp::yaml-cpp)
    set(YAML_TARGET yaml-cpp::yaml-cpp)
else()
    set(YAML_TARGET yaml-cpp)
endif()
configure_file("${CONTROLLER_SOURCE}" "${CMAKE_CURRENT_BINARY_DIR}/legacy_controller.c" COPYONLY)
configure_file("${CPP_CONTROLLER_SOURCE}" "${CMAKE_CURRENT_BINARY_DIR}/controller.cpp" COPYONLY)
if(MSVC)
    find_package(box2d CONFIG REQUIRED)
endif()
include(CTest)
foreach(language c cpp)
    set(target "installed_consumer_${language}")
    set(controller "${CMAKE_CURRENT_BINARY_DIR}/legacy_controller.c")
    if(language STREQUAL "cpp")
        set(controller "${CMAKE_CURRENT_BINARY_DIR}/controller.cpp")
    endif()
    add_executable(${target} "${controller}")
    set_target_properties(${target} PROPERTIES LINKER_LANGUAGE CXX)
    # Deliberately use only installed headers and the raw installed archive,
    # with the pre-coroutine dependency list and no explicit Boost.Context.
    target_include_directories(${target} PRIVATE "${INSTALL_PREFIX}/include")
    target_link_libraries(${target} PRIVATE "${INSTALLED_LIBRARY}"
        SDL2::SDL2 ${YAML_TARGET} spdlog::spdlog fmt::fmt
        ${SDL2_GFX_LIBRARIES} ${SDL2TTF_LIBRARY} Arrow::arrow_shared)
    if(MSVC)
        target_link_libraries(${target} PRIVATE box2d::box2d)
    else()
        if(APPLE)
            # Match legacy Makefiles: pinned Box2D lives in /usr/local, while
            # Homebrew libraries may live under /opt/homebrew on Apple Silicon.
            target_link_directories(${target} PRIVATE /usr/local/lib /opt/homebrew/lib)
        endif()
        target_link_libraries(${target} PRIVATE box2d)
    endif()
    add_test(NAME "installed_controller_${language}" COMMAND ${target} -c "${SIMULATION_CONFIG}" -g -nr)
    set_tests_properties("installed_controller_${language}" PROPERTIES WORKING_DIRECTORY "${SOURCE_DIR}"
        ENVIRONMENT "SDL_VIDEODRIVER=dummy" TIMEOUT 20)
endforeach()
